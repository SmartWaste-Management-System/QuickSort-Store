How the sort works

  1. Pick a pivot and move it to the end of the current range.
  2. Scan the range once. Every product that belongs before the pivot is swapped into the front part.
  3. Drop the pivot between the two parts. It is now in its final position.
  4. Repeat on the part to its left and the part to its right until every range has one item.
function quickSort(arr, lo = 0, hi = arr.length - 1) {
  if (lo >= hi) return;
  const p = partition(arr, lo, hi);
  quickSort(arr, lo, p - 1);
  quickSort(arr, p + 1, hi);
}

function partition(arr, lo, hi) {
  const pivot = arr[hi];
  let i = lo;
  for (let j = lo; j < hi; j++) {
    if (compare(arr[j], pivot) < 0) {
      swap(arr, i, j);
      i++;
    }
  }
  swap(arr, i, hi);
  return i;
}
